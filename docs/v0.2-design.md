### Blockers (must amend design before implementation)

| # | Issue | Fix |
|---|-------|-----|
| **B1** | Anonymizer salt would ship inside the bundle → defeats purpose | Salt lives in Keychain only (service `com.lucrativepictures.ModelStatus.anon`, account `salt`). Generated once at first bundle export. Never leaves device. |
| **B2** | sanitizeURL strips userinfo but not `?api_key=` / `?token=` query credentials | sanitizeURL drops entire query string by default. Bundle redaction scan adds query-param prefix match for `api_key`, `token`, `key`, `access_token`, `auth`. Document that ModelStatus does not support URL-string credentials. |
| **B3** | runShell "process-group kill via setpgid+killpg" not implementable via Foundation.Process | **Scope down**: design claims direct-child SIGKILL only. Always use explicit argument arrays (no `/bin/sh -c`) so no grandchild shell. Grandchild reaping deferred to v0.3 via posix_spawn helper. |

### Other fixes (implementation polish, not blockers)

- **SEC-M2**: processes.txt emits only `comm` + PID + port. Never raw `args` (could contain `--model /path/to/secret-project`).
- **SEC-M4**: `removeObserver(self)` at start of `applicationShouldTerminate`, not `applicationWillTerminate`.
- **L1**: memoize `isAppStoreInstalled()` at launch (`static let isAppStore: Bool`) — avoid filesystem stat on every menu open.
- **L2**: `walkMLXCache()` caps output at 200 entries, doesn't swallow errors with `try?`.
- **L3**: `README.txt` joins the bundle-wide redaction pass; test asserts hostname gets anonymized.
- **L4**: capability cache-miss fallback dispatches on `detectedKind`, not blanket `.openAI`.

- SEC-H1 (URL credential strip): closed by sanitizeURL + scrubURL, modulo B2.
- SEC-H2 (bundle-wide redaction): closed for 4 listed files; README.txt gap is L3.
- SEC-M1 (8-hex truncation): the security reviewer noted this is "acceptable for anonymization but not for proof-of-uniqueness." Original finding asked for full-length; we kept 8 hex (32 bits) intentionally because longer hashes are cosmetic noise. **Documented tradeoff, not a gap.**

## A. MLXProvider

### Detection strategy

MLX servers speak OpenAI-compat at `/v1/models`. The discriminating signal is the **model id pattern**:
- `mlx-community/*` prefix
- `*-4bit` / `*-8bit` / `*-bf16` suffix
- `owned_by == "mlx-omni"` (mlx-omni-server convention)

Rejection markers (route to OpenAIProvider/LMStudio instead):
- id contains `gguf` / `Q4_K_M` / `Q5_K_M` / `Q8_0`

For local instances, cross-confirm via `ps -axo args | grep mlx_lm.server` to avoid false claims on a generic OpenAI server that happens to serve an mlx-named model.

### MLXProvider.swift highlights

```swift
private struct MLXModelsResponse: Codable {
    struct M: Codable {
        let id: String
        let owned_by: String?
        let object: String?
    }
    let data: [M]?
}

struct MLXProvider: Provider {
    let kind: ProviderKind = .mlx
    let capabilities = ProviderCapabilities(
        canEject: false, canLoadModel: false,
        canListAvailable: true, reportsVRAM: true, reportsGenerating: true
    )

    func probe(_ instance: Instance, session: URLSession) async -> Bool {
        guard let base = URL(string: instance.url),
              let url = URL(string: "/v1/models", relativeTo: base) else { return false }
        do {
            let (data, http, _) = try await HTTPHelpers.get(url, instanceID: instance.id,
                                                            session: session, timeout: 3)
            guard http.statusCode == 200,
                  let resp = try? JSONDecoder().decode(MLXModelsResponse.self, from: data),
                  let models = resp.data, !models.isEmpty else { return false }

            if !models.contains(where: Self.idLooksLikeMLX) { return false }
            if models.contains(where: Self.idLooksLikeGGUF) { return false }

            if LocalProbe.isLocal(instance.url) {
                return await Self.localProcessIsMLX()
            }
            return true
        } catch { return false }
    }

    static func idLooksLikeMLX(_ m: MLXModelsResponse.M) -> Bool {
        if m.owned_by?.lowercased() == "mlx-omni" { return true }
        if m.id.lowercased().hasPrefix("mlx-community/") { return true }
        if m.id.range(of: #"-(4|8)bit$"#, options: .regularExpression) != nil { return true }
        if m.id.hasSuffix("-bf16") { return true }
        return false
    }

    static func idLooksLikeGGUF(_ m: MLXModelsResponse.M) -> Bool {
        let l = m.id.lowercased()
        return l.contains("gguf") || l.contains("q4_k_m") || l.contains("q5_k_m") || l.contains("q8_0")
    }

    static func walkMLXCache() -> [String] {
        let hub = FileManager.default.homeDirectoryForCurrentUser
            .appendingPathComponent(".cache/huggingface/hub")
        var isDir: ObjCBool = false
        guard FileManager.default.fileExists(atPath: hub.path, isDirectory: &isDir),
              isDir.boolValue,
              let entries = try? FileManager.default.contentsOfDirectory(atPath: hub.path)
        else { return [] }
        let prefix = "models--mlx-community--"
        return entries.compactMap { name -> String? in
            guard name.hasPrefix(prefix) else { return nil }
            return String(name.dropFirst("models--".count))
                .replacingOccurrences(of: "--", with: "/")
        }
    }

    static func rssBytes(pid: Int) async -> Int64 {
        guard pid > 0,
              let out = await LocalProbe.runShell("/bin/ps",
                                                  args: ["-p", String(pid), "-o", "rss="]) else { return 0 }
        let kb = Int64(out.trimmingCharacters(in: .whitespacesAndNewlines)
            .split(whereSeparator: { $0.isWhitespace }).first.map(String.init) ?? "") ?? 0
        return kb * 1024
    }
}
```

## D. runShell timeout race hardening

### Root cause

`readDataToEndOfFile()` parks the worker thread indefinitely if the child traps SIGTERM and keeps stdout open.

### Fix pattern

1. Stream reads via `readabilityHandler` + accumulator buffer (no blocking read).
2. Two-stage kill: SIGTERM at `T+timeout`, SIGKILL at `T+timeout+2`.
3. Single-fire latch (DispatchQueue-guarded Bool — no atomics dep) so multiple resolution paths (natural exit, SIGTERM-exit, SIGKILL-exit) can race to finish, with only one continuation.resume.

### Code

```swift
static func runShell(_ path: String, args: [String], timeout: TimeInterval = 5) async -> String? {
    await withCheckedContinuation { (continuation: CheckedContinuation<String?, Never>) in
        let proc = Process()
        proc.executableURL = URL(fileURLWithPath: path)
        proc.arguments = args
        let pipe = Pipe()
        proc.standardOutput = pipe
        proc.standardError = FileHandle.nullDevice

        let sync = DispatchQueue(label: "modelstatus.runshell.sync")
        var buffer = Data()
        var resumed = false
        let readFH = pipe.fileHandleForReading

        readFH.readabilityHandler = { fh in
            let chunk = fh.availableData
            if chunk.isEmpty { fh.readabilityHandler = nil; return }
            sync.sync { buffer.append(chunk) }
        }

        let tryFinish: (Bool) -> Void = { _ in
            let shouldResume = sync.sync { () -> Bool in
                if resumed { return false }
                resumed = true; return true
            }
            guard shouldResume else { return }
            readFH.readabilityHandler = nil
            let tail = (try? readFH.readToEnd()) ?? Data()
            let final: Data = sync.sync { buffer.append(tail); return buffer }
            try? readFH.close()
            continuation.resume(returning: String(data: final, encoding: .utf8))
        }

        proc.terminationHandler = { _ in tryFinish(false) }

        do { try proc.run() } catch {
            readFH.readabilityHandler = nil
            try? readFH.close()
            continuation.resume(returning: nil)
            return
        }

        let killQueue = DispatchQueue.global(qos: .utility)
        killQueue.asyncAfter(deadline: .now() + timeout) {
            if proc.isRunning { proc.terminate() }    // SIGTERM
        }
        killQueue.asyncAfter(deadline: .now() + timeout + 2.0) {
            if proc.isRunning { kill(proc.processIdentifier, SIGKILL) }
        }
    }
}
```

### LogViewerWindow

`OSLogStore(scope: .currentProcessIdentifier)` — no entitlement needed, runs sandbox-clean. NSPredicate filter on subsystem + optional category. Auto-refresh every 2s when window visible.

Key bits:
```swift
private static func fetchEntries(categoryFilter: String) async throws -> [OSLogEntryLog] {
    try await Task.detached(priority: .userInitiated) {
        let store = try OSLogStore(scope: .currentProcessIdentifier)
        let start = store.position(date: Date().addingTimeInterval(-3600))   // 1 hour
        let subsystem = ConfigManager.bundleIdentifier
        let predicate = categoryFilter == "All"
            ? NSPredicate(format: "subsystem == %@", subsystem)
            : NSPredicate(format: "subsystem == %@ AND category == %@", subsystem, categoryFilter)
        let raw = try store.getEntries(at: start, matching: predicate)
        var out: [OSLogEntryLog] = []
        for entry in raw {
            guard let log = entry as? OSLogEntryLog else { continue }
            out.append(log)
            if out.count >= 1000 { break }
        }
        return out
    }.value
}
```
